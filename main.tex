% \pgffigure{<key>} inserts a float figure from figures/<key>.pgf, captions it
% from captions/<key>.tex, and labels it fig:<key>.
\NewDocumentCommand{\pgffigure}{s O{htbp} m}{%
  \IfBooleanTF{#1}{%
    \begin{figure*}[#2]%
      \centering%
      \input{figures/#3.pgf}%
      \caption{\protect\input{captions/#3.tex}}%
      \label{fig:#3}%
    \end{figure*}%
  }{%
    \begin{figure}[#2]%
      \centering%
      \input{figures/#3.pgf}%
      \caption{\protect\input{captions/#3.tex}}%
      \label{fig:#3}%
    \end{figure}%
  }%
}
